                FROM    Orders O
                WHERE   O.shippedDate IS NULL;

                #   Fetches all the data from the Orders database query and abbriviates the name as O

            c)  SELECT  C.customerName AS Customer, SUM(OD.quantityOrdered) AS Total
                FROM    Orders O, Customers C, OrderDetails OD
                WHERE   O.customerNumber = C.customerNumber
                        AND     O.orderNumber = OD.orderNumber
                GROUP BY    O.customerNumber
                ORDER BY    Total DESC;

                //  Fetches the costumers who made a spesific order and sorts them by how much they ordered
                #   Fetches customer names from costumers query to the alias customer,
                    fetches the sum of the items in the quantity ordered attribute in the order details query to the alias Total
                    and sets the orders, costumers and order details database queries to an abbriviated character.
                #   Filters the results to only include results where the costumer number attribute is the same from the orders query and the costumers query
                    and results where the order number attribute is the same in the orders query and the order details query
                #   Arranges the identical customer numbers from the orders query into groups
                #   Sort the results in a descending numerical order using the alias total
                
            d)  SELECT  P.productName, T.totalQuantityOrdered
                FROM    Products P NATURAL JOIN
                        (SELECT productCode, SUM(quantityOrdered) AS totalQuantityOrdered
                        FROM    OrderDetails GROUP BY productCode) AS T
                WHERE   T.totalQuantityOrdered >= 1000;

                //  Fetches the products whose collected orders are larger in quantity than 1000.
                #   Fetches the product names from the Products database query abbriviated as P,
                    fetches the sum of the quantity ordered attribute based on the product code and sets it as the alias total quantity ordered
                    specifies the order details query to be grouped by the product code and naturally joined as the alias T
                #   Filters the results to only list results where the total quantity ordered attribute from the alias T is bigger than equal to 1000
